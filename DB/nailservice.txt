-- Создаем базу данных
CREATE DATABASE IF NOT EXISTS nailservicedb;
USE nailservicedb;

-- Таблица ролей
CREATE TABLE IF NOT EXISTS role (
    IDRole INT PRIMARY KEY AUTO_INCREMENT,
    RoleName VARCHAR(50) NOT NULL UNIQUE
);

-- Таблица статусов
CREATE TABLE IF NOT EXISTS status (
    IDStatus INT PRIMARY KEY AUTO_INCREMENT,
    StatusName VARCHAR(50) NOT NULL UNIQUE
);

-- Таблица пользователей (с IsActive для soft delete)
CREATE TABLE IF NOT EXISTS users (
    IDUser INT PRIMARY KEY AUTO_INCREMENT,
    LastName VARCHAR(50) NOT NULL,
    FirstName VARCHAR(50) NOT NULL,
    MiddleName VARCHAR(50) DEFAULT NULL,
    Login VARCHAR(50) NOT NULL UNIQUE,
    Password VARCHAR(255) NOT NULL,
    Role INT NOT NULL,
    IsActive BOOLEAN NOT NULL DEFAULT TRUE,
    FOREIGN KEY (Role) REFERENCES role(IDRole)
);

-- Таблица категорий услуг (с IsActive для soft delete)
CREATE TABLE IF NOT EXISTS category (
    IDCategory INT PRIMARY KEY AUTO_INCREMENT,
    CategoryName VARCHAR(100) NOT NULL,
    IsActive BOOLEAN NOT NULL DEFAULT TRUE
);

-- Таблица услуг (с IsActive для soft delete)
CREATE TABLE IF NOT EXISTS services (
    IDServices INT PRIMARY KEY AUTO_INCREMENT,
    ServiceName VARCHAR(100) NOT NULL,
    Description TEXT,
    Price DECIMAL(10,2) NOT NULL,
    Photo VARCHAR(255) DEFAULT NULL,
    Category INT NOT NULL,
    IsActive BOOLEAN NOT NULL DEFAULT TRUE,
    FOREIGN KEY (Category) REFERENCES category(IDCategory)
);

-- Таблица клиентов (с IsActive для soft delete)
CREATE TABLE IF NOT EXISTS client (
    IDClient INT PRIMARY KEY AUTO_INCREMENT,
    LastName VARCHAR(50) NOT NULL,
    FirstName VARCHAR(50) NOT NULL,
    MiddleName VARCHAR(50) DEFAULT NULL,
    Phone VARCHAR(20) NOT NULL,
    IsActive BOOLEAN NOT NULL DEFAULT TRUE
);

-- Таблица мастеров (с IsActive для soft delete)
CREATE TABLE IF NOT EXISTS masters (
    IDMasters INT PRIMARY KEY AUTO_INCREMENT,
    User INT NOT NULL,
    Description TEXT,
    Phone VARCHAR(20) NOT NULL,
    IsActive BOOLEAN NOT NULL DEFAULT TRUE,
    FOREIGN KEY (User) REFERENCES users(IDUser)
);

-- Таблица записей (это история, не удаляем)
CREATE TABLE IF NOT EXISTS record (
    IDRecord INT PRIMARY KEY AUTO_INCREMENT,
    Master INT NOT NULL,
    Client INT NOT NULL,
    Date DATETIME NOT NULL,
    Status INT NOT NULL,
    Service INT NOT NULL,
    User INT NOT NULL,
    FOREIGN KEY (Master) REFERENCES masters(IDMasters),
    FOREIGN KEY (Client) REFERENCES client(IDClient),
    FOREIGN KEY (Status) REFERENCES status(IDStatus),
    FOREIGN KEY (Service) REFERENCES services(IDServices),
    FOREIGN KEY (User) REFERENCES users(IDUser)
);
